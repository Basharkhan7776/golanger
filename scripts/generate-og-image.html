  "github.com/gin-gonic/gin"
)

  r.GET("/api/learn", func(c *gin.Context) {
    c.JSON(200, gin.H{
      "course": "Golanger",
      "status": "concurrency-ready",
    })
  })
  r.Run(":8080")
}
